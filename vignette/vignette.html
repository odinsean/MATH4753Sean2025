myboot2

Compute a 95% bootstrap CI for the median fire damage

set.seed(123)
boot_result <- myboot2(iter = 1000, x = fire$DAMAGE, fun = median, alpha = 0.05)

boot_result$ci
#>  2.5% 97.5% 
#>  22.3  31.3

mymlnorm

Fit a normal distribution

ml_result <- mymlnorm(
  x = fire$DAMAGE,
  mu = seq(10, 25, length = 100),
  sig = seq(3, 8, length = 100)
)

myncurve

Plot a normal curve with mean/sd and shade P(X ≤ 20)

myncurve(
  mu = mean(fire$DAMAGE),
  sigma = sd(fire$DAMAGE),
  a = 20
